# Week 4 - Interactive Generative Poster
# Concepts: sliders, interactivity, parameter control

import random
import math
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.colors import hsv_to_rgb

import ipywidgets as widgets
from ipywidgets import interact


# --------------------------------------------------
# 1. Blob 도형 생성 함수
# --------------------------------------------------

def blob(center=(0.5, 0.5), r=0.3, points=200, wobble=0.15):

    # 원을 구성할 각도 생성
    angles = np.linspace(
        0,
        2 * math.pi,
        points,
        endpoint=False
    )

    # 반지름에 랜덤한 변화를 주어
    # 울퉁불퉁한 유기적 형태 생성
    radii = r * (
        1 + wobble * (np.random.rand(points) - 0.5)
    )

    # 극좌표를 x, y 좌표로 변환
    x = center[0] + radii * np.cos(angles)
    y = center[1] + radii * np.sin(angles)

    return x, y


# --------------------------------------------------
# 2. 색상 팔레트 생성 함수
# --------------------------------------------------

def make_palette(k=6, mode="pastel", base_h=0.60):

    colors = []

    for _ in range(k):

        # 파스텔 색상
        if mode == "pastel":

            h = random.random()
            s = random.uniform(0.15, 0.35)
            v = random.uniform(0.9, 1.0)

        # 강렬한 색상
        elif mode == "vivid":

            h = random.random()
            s = random.uniform(0.8, 1.0)
            v = random.uniform(0.8, 1.0)

        # 단색 계열
        elif mode == "mono":

            h = base_h
            s = random.uniform(0.2, 0.6)
            v = random.uniform(0.5, 1.0)

        # 완전 랜덤
        else:

            h = random.random()
            s = random.uniform(0.3, 1.0)
            v = random.uniform(0.5, 1.0)

        rgb = tuple(
            hsv_to_rgb([h, s, v])
        )

        colors.append(rgb)

    return colors


# --------------------------------------------------
# 3. 포스터 생성 함수
# --------------------------------------------------

def draw_poster(
    n_layers=8,
    wobble=0.15,
    palette_mode="pastel",
    seed=0
):

    # 같은 Seed를 사용하면
    # 같은 포스터가 나오도록 설정
    random.seed(seed)
    np.random.seed(seed)

    # 포스터 비율
    fig, ax = plt.subplots(
        figsize=(6, 8)
    )

    # 축 숨기기
    ax.axis("off")

    # 약간 따뜻한 흰색 배경
    ax.set_facecolor(
        (0.97, 0.97, 0.97)
    )

    # 팔레트 생성
    palette = make_palette(
        6,
        mode=palette_mode
    )

    # 여러 개의 Blob 생성
    for _ in range(n_layers):

        # 랜덤 위치
        cx = random.random()
        cy = random.random()

        # 랜덤 크기
        radius = random.uniform(
            0.15,
            0.45
        )

        # Blob 생성
        x, y = blob(
            center=(cx, cy),
            r=radius,
            wobble=wobble
        )

        # 랜덤 색상
        color = random.choice(
            palette
        )

        # 반투명도
        alpha = random.uniform(
            0.3,
            0.6
        )

        # 도형 그리기
        ax.fill(
            x,
            y,
            color=color,
            alpha=alpha,
            edgecolor=(0, 0, 0, 0)
        )

    # 포스터 제목
    ax.text(
        0.05,
        0.95,
        f"Interactive Poster • {palette_mode}",
        transform=ax.transAxes,
        fontsize=12,
        weight="bold"
    )

    plt.show()


# --------------------------------------------------
# 4. 인터랙티브 UI
# --------------------------------------------------

interact(

    draw_poster,

    n_layers=widgets.IntSlider(
        min=3,
        max=20,
        step=1,
        value=8,
        description="Layers"
    ),

    wobble=widgets.FloatSlider(
        min=0.01,
        max=0.30,
        step=0.01,
        value=0.15,
        description="Wobble"
    ),

    palette_mode=widgets.Dropdown(
        options=[
            "pastel",
            "vivid",
            "mono",
            "random"
        ],
        value="pastel",
        description="Palette"
    ),

    seed=widgets.IntSlider(
        min=0,
        max=9999,
        step=1,
        value=0,
        description="Seed"
    )
)
